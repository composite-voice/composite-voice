class MockLLM extends BaseLLMProvider {
  constructor(config = {}) {
    super({
      systemPrompt: config.systemPrompt || 'You are a helpful assistant.',
      maxTokens: config.maxTokens || 200,
      temperature: config.temperature || 0.7,
      ...config,
    });
    this.responses = config.responses || [
      "I'm a mock LLM running entirely in your browser!",
      "This shows how to extend BaseLLMProvider.",
      "You could connect this to any LLM backend.",
      "The SDK doesn't care where the text comes from.",
      "Try saying different things! I cycle through responses.",
    ];
    this.responseIndex = 0;
  }

  async onInitialize() {
    // Nothing to initialize
  }

  async onDispose() {
    // Nothing to clean up
  }

  async generate(prompt, options) {
    return this.generateFromMessages(
      [{ role: 'user', content: prompt }], options
    );
  }

  async generateFromMessages(messages, options) {
    this.assertReady();
    const response = this.responses[
      this.responseIndex++ % this.responses.length
    ];

    return {
      async *[Symbol.asyncIterator]() {
        const words = response.split(' ');
        for (const word of words) {
          if (options?.signal?.aborted) break;
          yield word + ' ';
          await new Promise(r => setTimeout(r, tokenDelay));
        }
      },
    };
  }
}
